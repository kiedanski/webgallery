<project_specification>
  <project_name>AVU - Asociación Vegana del Uruguay - Member Management Web App</project_name>

  <overview>
    Build a fully functional clone of the Asociación Vegana del Uruguay (AVU) web app at avu.org.uy.
    The application serves as the public-facing website and member management platform for a Uruguayan
    vegan NGO. It combines a clean, mostly-static public site (home, about, FAQ, benefits, contact)
    with a full membership lifecycle: registration, payment (MercadoPago or wire transfer), benefit
    access, and email communications. A custom (non-technical-friendly) admin dashboard handles member
    approval, benefit management, campaign sending, and usage statistics. The site is in Spanish (es-UY),
    responsive by default, and optimized for mobile. Built with Django + htmx for a snappy, low-JS feel.
  </overview>

  <technology_stack>

      <framework>Django 5.x</framework>
      <database>PostgreSQL</database>
      <dynamic_ui>htmx for partial page updates (no full SPA)</dynamic_ui>

  <core_features>

    <public_site>
      - Home page: hero section, brief about AVU, call-to-action to join
      - Sobre el Veganismo (About Veganism): educational static content
      - Preguntas Frecuentes (FAQ): accordion-style Q&amp;A, static
      - Beneficios (Benefits): grid of partner businesses with discount info; map view for those with addresses
      - Contacto (Contact): form with name, email, subject, message — submits via backend POST to CONTACT_EMAIL (no mailto:)
      - Footer: social links (Facebook, Instagram), email, legal info
      - Navigation: mobile hamburger menu, collapsible on small screens
    </public_site>

    <membership_registration>
      - Registration form collects:
        - Nombre (first name)
        - Apellido (last name)
        - Correo electrónico (email, used as login)
        - Contraseña (password, confirmed)
        - Cédula de Identidad (stored as plain text, no external validation)
        - Dirección (street address, city, department)
        - Teléfono (phone number)
      - Account created with status = PENDING (inactive, no benefit access)
      - After registration, user is redirected to payment selection page
      - One active subscription per user at a time
      - Users can log in but see a "subscription inactive" banner until paid
    </membership_registration>

    <payment_flows>
      <mercadopago>
        - User selects "Pagar con MercadoPago" on payment page
        - Django backend creates MercadoPago preference via SDK
        - User redirected to MercadoPago hosted checkout (redirect flow, works on mobile)
        - MercadoPago calls webhook on payment success/failure/pending
        - Webhook endpoint verifies signature, updates subscription status to ACTIVE
        - Subscription expiry set to 30 days from payment date
        - Monthly recurring: MercadoPago sends new webhook each billing cycle
        - On failed renewal: status set to EXPIRED, automated expiry warning email sent
      </mercadopago>

      <wire_transfer>
        - User selects "Transferencia bancaria" on payment page
        - Page shows AVU bank account details (BROU, account number, etc.) and instructions
        - User fills upload form: full name, amount paid, date of transfer, PDF proof file
        - PDF uploaded to local media storage (or Railway volume)
        - LlamaExtract API called to extract: payee name, amount, date from PDF
        - Extracted data stored alongside original upload for admin review
        - Submission creates WireTransferRequest with status = PENDING_REVIEW
        - Admin sees pending requests in dashboard with extracted data + original PDF link
        - Admin can APPROVE (activates subscription) or REJECT (with optional note sent to user by email)
      </wire_transfer>

      <activation_code>
        - Admin can generate single-use alphanumeric activation codes in dashboard
        - Code can optionally have an expiry date and a note (e.g. "Gift membership - Ana")
        - User enters code on payment page under "Tengo un código de activación"
        - Valid unused code: subscription activated immediately, code marked used
        - Invalid/expired/used code: inline htmx error shown without page reload
      </activation_code>
    </payment_flows>

    <subscription_lifecycle>
      - Statuses: PENDING | ACTIVE | EXPIRED | SUSPENDED
      - Subscription record tracks: user, payment_method, start_date, expiry_date, status
      - 7 days before expiry: automated email sent warning of upcoming expiration
      - On expiry: status set to EXPIRED, benefit access revoked immediately
      - User can renew from their member dashboard
      - Renewal creates new subscription record (history preserved)
    </subscription_lifecycle>

    <member_area>
      - Accessible only to authenticated users with ACTIVE subscription
      - Dashboard shows: subscription status, expiry date, payment method, renewal button
      - Benefits page (same as public but with full discount details visible)
      - Payment history: list of past subscriptions and payment method used
      - Profile page: view and edit address, phone, password (cédula not editable after registration)
    </member_area>

    <benefits_system>
      - Benefit fields: name, description, discount_details, category, website_url (optional),
        address (optional), city, latitude, longitude (optional, for map), logo image, is_active
      - Categories: Restaurante, Tienda, Servicio, Salud, Otro
      - Public benefits page: grid of active benefits, filterable by category via htmx
      - Map view tab: Leaflet.js map showing pins for benefits that have coordinates
      - Clicking a pin shows benefit name + discount in popup
      - Benefits without addresses appear only in the grid, not on map
      - Admin can geocode address to lat/lng manually or enter coordinates directly
    </benefits_system>

    <email_communications>
      - Admin can compose and send email campaigns to all active members
      - Compose form: subject, body (rich text via simple textarea with markdown-style help)
      - Preview rendered email before sending
      - Send immediately to all active members via Resend API
      - No scheduling — send now only
      - Campaign log: list of past campaigns with sent_at, subject, recipient_count
      - Automated system emails (non-campaign):
        - Welcome email on registration
        - Payment confirmation on activation
        - 7-day expiry warning
        - Subscription expired notification
        - Wire transfer rejection email (with admin note)
        - Wire transfer approval email
      - All emails sent from a configured FROM address (e.g. socios@avu.org.uy via Resend)
    </email_communications>

    <contact_form>
      - Fields: nombre, email, asunto (subject), mensaje (message)
      - On submit: Django sends email to CONTACT_EMAIL via Resend
      - htmx: form replaced with success message inline without page reload
      - Rate-limited: max 3 submissions per IP per hour (Django middleware)
      - No CAPTCHA required (rate limiting sufficient for low-volume NGO)
    </contact_form>

    <admin_dashboard>
      - Separate from Django's built-in admin (which is still available for superusers)
      - Accessible at /dashboard/ — requires staff user login
      - Designed for non-technical administrators (clear labels, confirmations, no jargon)

      <stats_overview>
        - Total active members count
        - New members this month
        - Members by payment method (MercadoPago vs Wire Transfer vs Code)
        - Expiring soon (next 7 days) count
        - Total revenue estimate this month (active * fee)
      </stats_overview>

      <member_management>
        - Paginated member list: name, email, cédula, status, payment method, expiry date
        - Search by name, email, or cédula
        - Filter by status (active, pending, expired)
        - Click member to view full profile and subscription history
        - Admin can manually change subscription status (with audit log note)
        - Admin can trigger expiry warning email manually
      </member_management>

      <wire_transfer_review>
        - Queue of pending wire transfer requests
        - Each shows: submitter name, submitted amount, extracted amount (LlamaExtract), date, PDF link
        - Approve button: activates subscription, sends confirmation email
        - Reject button: prompts for rejection note, sends rejection email
        - Approved/rejected history viewable
      </wire_transfer_review>

      <activation_codes>
        - Generate new code: optionally set expiry date and internal note
        - List all codes: code, created_by, expiry, used_by (if used), used_at
        - Deactivate unused code
      </activation_codes>

      <benefit_management>
        - CRUD for benefits (create, edit, deactivate, delete)
        - Image upload for benefit logo
        - Address + optional manual lat/lng input
        - Preview benefit card inline
        - Toggle active/inactive without deleting
      </benefit_management>

      <campaign_composer>
        - Compose email: subject + body textarea
        - Live preview of rendered email (htmx partial)
        - Send button with confirmation modal (shows recipient count)
        - Campaign history list
      </campaign_composer>
    </admin_dashboard>

  </core_features>

  <database_schema>
    <tables>
      <users>
        - id, email (unique, used as username), first_name, last_name
        - cedula (CharField, plain text)
        - address, city, department, phone
        - date_joined, last_login
        - is_active, is_staff (Django defaults)
        - custom_instructions (not used, but Django AbstractUser compatible)
      </users>

      <subscriptions>
        - id, user_id (FK)
        - status: PENDING | ACTIVE | EXPIRED | SUSPENDED
        - payment_method: MERCADOPAGO | WIRE_TRANSFER | ACTIVATION_CODE | MANUAL
        - start_date, expiry_date
        - mercadopago_subscription_id (nullable)
        - activation_code_id (FK, nullable)
        - created_at, updated_at
      </subscriptions>

      <wire_transfer_requests>
        - id, user_id (FK)
        - pdf_file (FileField)
        - submitted_amount (DecimalField)
        - submitted_date (DateField)
        - submitter_name (CharField)
        - extracted_payee, extracted_amount, extracted_date (from LlamaExtract, nullable)
        - status: PENDING_REVIEW | APPROVED | REJECTED
        - reviewed_by (FK to User, nullable)
        - reviewed_at (nullable)
        - rejection_note (TextField, nullable)
        - created_at
      </wire_transfer_requests>

      <activation_codes>
        - id, code (CharField, unique, alphanumeric)
        - created_by (FK to User)
        - note (CharField, internal only)
        - expires_at (nullable)
        - used_by (FK to User, nullable)
        - used_at (nullable)
        - is_active (BooleanField)
        - created_at
      </activation_codes>

      <benefits>
        - id, name, description, discount_details
        - category: RESTAURANTE | TIENDA | SERVICIO | SALUD | OTRO
        - website_url (nullable)
        - address, city (nullable)
        - latitude, longitude (DecimalField, nullable)
        - logo (ImageField, nullable)
        - is_active (BooleanField)
        - created_at, updated_at
      </benefits>

      <email_campaigns>
        - id, subject, body
        - sent_by (FK to User)
        - sent_at
        - recipient_count
      </email_campaigns>

      <contact_submissions>
        - id, name, email, subject, message
        - ip_address
        - created_at
        - email_sent (BooleanField)
      </contact_submissions>
    </tables>
  </database_schema>
